// Helper Functions for P21 and P22
int addNumbers(int n1, int n2) {
  return n1 + n2;
}

bool isPrime(int n) {
  if (n <= 1) return false;
  for (int i = 2; i <= n ~/ 2; i++) {
    if (n % i == 0) return false;
  }
  return true;
}
