class LabSheet2App extends StatelessWidget {

    {'title': '6. Text, Image & Icon Combo', 'widget': Program6()},

    {'title': '10. Three Buttons UI', 'widget': Program10()},
    {'title': '11. Student Profile Card', 'widget': Program11()},
    {'title': '12. Business Card UI', 'widget': Program12()},
    {'title': '13. Simple Login Screen', 'widget': Program13()},
    {'title': '14. Registration Screen', 'widget': Program14()},
    {'title': '15. College Home Screen', 'widget': Program15()},
    {'title': '16. Products with Cards', 'widget': Program16()},
    {'title': '17. Student Info Card', 'widget': Program17()},
    {'title': '18. Padding & Margin Demo', 'widget': Program18()},
    {'title': '19. Center & Align Widgets', 'widget': Program19()},
    {'title': '20. Food Menu Screen', 'widget': Program20()},
    {'title': '21. Calculator Interface', 'widget': Program21()},
    {'title': '22. Weather Info Screen', 'widget': Program22()},
    {'title': '23. Shopping Product Screen', 'widget': Program23()},
    {'title': '24. Student Dashboard', 'widget': Program24()},
    {'title': '25. Complete Personal Profile', 'widget': Program25()},
  ];

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Lab Sheet - 2 Solutions'),
        backgroundColor: Colors.teal,
        foregroundColor: Colors.white,
      ),
      body: ListView.builder(
        itemCount: programs.length,
        itemBuilder: (context, index) {
          return Card(
            margin: const EdgeInsets.symmetric(horizontal: 10, vertical: 4),
            child: ListTile(
              leading: CircleAvatar(
                backgroundColor: Colors.teal,
                foregroundColor: Colors.white,
                child: Text('${index + 1}'),
              ),
              title: Text(programs[index]['title']),
              trailing: const Icon(Icons.arrow_forward_ios, size: 16),
              onTap: () {
                Navigator.push(
                  context,
                  MaterialPageRoute(
                    builder: (context) => Scaffold(
                      appBar: AppBar(
                        title: Text(programs[index]['title']),
                        backgroundColor: Colors.teal,
                        foregroundColor: Colors.white,
                      ),
                      body: Padding(
                        padding: const EdgeInsets.all(16.0),
                        child: programs[index]['widget'],
                      ),
                    ),
                  ),
                );
              },
            ),
          );
        },
      ),
    );
  }
}
