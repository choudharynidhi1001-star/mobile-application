3. Arithmetic Operations (+, -, *, /)
--------------------------------------------------
class Program3 extends StatefulWidget {
  const Program3({super.key});
  @override
  State<Program3> createState() => _Program3State();
}
class _Program3State extends State<Program3> {
  final _n1 = TextEditingController(), _n2 = TextEditingController();
  String _res = '';
  void _calc(String op) {
    double a = double.tryParse(_n1.text) ?? 0, b = double.tryParse(_n2.text) ?? 0;
    setState(() {
      if (op == '+') _res = 'Result: ${a + b}';
      if (op == '-') _res = 'Result: ${a - b}';
      if (op == '*') _res = 'Result: ${a * b}';
      if (op == '/') _res = b != 0 ? 'Result: ${a / b}' : 'Cannot divide by zero';
    });
  }
  @override
  Widget build(BuildContext context) => Column(
    children: [
      TextField(controller: _n1, keyboardType: TextInputType.number, decoration: const InputDecoration(labelText: 'Number 1')),
      TextField(controller: _n2, keyboardType: TextInputType.number, decoration: const InputDecoration(labelText: 'Number 2')),
      Row(
        mainAxisAlignment: MainAxisAlignment.spaceEvenly,
        children: [
          ElevatedButton(onPressed: () => _calc('+'), child: const Text('+')),
          ElevatedButton(onPressed: () => _calc('-'), child: const Text('-')),
          ElevatedButton(onPressed: () => _calc('*'), child: const Text('*')),
          ElevatedButton(onPressed: () => _calc('/'), child: const Text('/')),
        ],
      ),
      Text(_res, style: const TextStyle(fontSize: 18)),
    ],
  );
}
